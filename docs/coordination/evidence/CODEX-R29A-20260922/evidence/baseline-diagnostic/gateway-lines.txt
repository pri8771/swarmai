   320	        if terminal is None:
   321	            raise ReconciliationRequiredError("succeeded_effect_missing_receipt")
   322	        return terminal

   324	    def _finalize(
   325	        self,
   326	        envelope: ActionEnvelope,
   327	        effect: dict[str, Any],
   328	        *,
   329	        state: str,
   330	        outcome: str,
   331	        pre: dict[str, Any],
   332	        post: dict[str, Any],
   333	        started: Any = None,
   334	        external_id: str | None = None,
   335	        reconciliation: dict[str, Any] | None = None,
   336	        reconciliation_state: str = "none",
   337	        artifacts: list[str] | None = None,
   338	        state_reason: str | None = None,
   339	    ) -> ActionReceiptV17:
   340	        receipt = self._build_receipt(
   341	            envelope,
   342	            effect_id=effect["effect_id"],
   343	            outcome=outcome,
   344	            pre=pre,
   345	            post=post,
   346	            started=started,
   347	            reconciliation_state=reconciliation_state,
   348	            external_id=external_id,
   349	            artifacts=artifacts,
   350	        )
   351	        return self.store.finalize_with_receipt(
   352	            project_id=envelope.project_id,
   353	            effect_key=envelope.effect_key,
   354	            state=state,
   355	            state_reason=state_reason,
   356	            post_observation=post,
   357	            external_id=external_id,
   358	            reconciliation=reconciliation,
   359	            receipt=receipt,
   360	            expected_execution=(
   361	                effect.get("executor_id"),
   362	                effect["attempt_count"],
   363	                effect["state"],
   364	            ),
   365	        )
   600	        arts: list[str] = []
   601	        result = post.get("result") if isinstance(post.get("result"), dict) else post
   602	        if isinstance(result, dict):
   603	            if result.get("path"):
   604	                arts.append(f"file:{result['path']}")
   605	            if result.get("external_id"):
   606	                arts.append(f"ext:{result['external_id']}")
   607	            if result.get("navigated_to"):
   608	                arts.append(f"url:{result['navigated_to']}")
   609	        return arts
