   795	            )

   800	    def get_receipt(self, action_id: str) -> ActionReceiptV17 | None:
   801	        with session_scope(self.factory) as session:
   802	            row = session.scalar(
   803	                select(ActionReceiptRow)
   804	                .where(ActionReceiptRow.action_id == action_id)
   805	                .order_by(
   806	                    ActionReceiptRow.created_at.desc(), ActionReceiptRow.attempt_number.desc()
   807	                )
   808	                .limit(1)
   809	            )
